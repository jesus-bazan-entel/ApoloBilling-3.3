    id, username, action, entity_type, entity_id,
    jsonb_pretty(details) as detalles,
    created_at

-- Ver todas las modificaciones de una tarifa específica

       details->>'destination_prefix' as prefix,
       created_at
FROM audit_logs
WHERE entity_type = 'rate_card'
  AND entity_id = '19'
ORDER BY created_at;

-- Ver cambios en la última hora
SELECT id, username, action, entity_type, created_at
FROM audit_logs
WHERE created_at >= NOW() - INTERVAL '1 hour'
  AND username != 'postgres'
ORDER BY created_at DESC;
